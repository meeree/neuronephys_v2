226 __device__ void dxdt_an_ie_model(const int idx,
227  Float const V, Float const nK, Float const Ca, Float const Ye, Float const Yi, Float const Y_loc,
228  Float const e_Tinavg, Float const i_Tinavg, Float const loc_Tinavg,
229  Float const white_noise,
230  Float* dvdt, Float* dnKdt, Float* dCadt, Float* dYedt, Float* dYidt, Float* dY_loc_dt)
231 {
232  using namespace SAN;
233  // Calculate intermediate quantities
234  Float mCa_inf = 1 / (1 + exp(-(V + 20) / 9));
235  Float mKCa_inf = 1 / (1 + powf(kD / Ca, 3.5));
236  Float mNaP_inf = 1 / (1 + exp(-(V + 55.7) / 7.7));
237 
238  Float alpha_n = V == -34 ? 0.1 : 0.01 * (V + 34) / (1 - exp(-(V + 34) / 10));
239  Float beta_n = 0.125 * exp(-(V + 44) / 25);
240 
241  // Calculate gating variable derivatives
242  *dnKdt = 4 * (alpha_n * (1 - nK) - beta_n * nK);
243  *dYedt = 3.48 * e_Tinavg - Ye / tauAMPA_white;
244  *dY_loc_dt = 3.48 * loc_Tinavg - Y_loc / tauAMPA_grey;
245  *dYidt = i_Tinavg - Yi / tauGABA;
246 
247  // Calculate Ca2+ derivative
248  Float ICa = gCa * mCa_inf * mCa_inf * (V - vCa);
249  *dCadt = -alphaCa * (10 * A * ICa) - Ca / tauCa;
250 
251  // Calculate voltage derivative
252  *dvdt =
253  white_noise
254  - gK * powf(nK, 4) * (V - vK)
255  - ICa
256  - gKCa * mKCa_inf * (V - vK)
257  - gNaP * mNaP_inf * mNaP_inf * mNaP_inf * (V - vNa)
258  - gL * (V - vL)
259  - gAMPA_white * Ye * (V - vAMPA)
260  - gAMPA_grey * Y_loc * (V - vGABA)
261  - gGABA * Yi * (V - vGABA);
262 
263  *dvdt /= C;
264 }
