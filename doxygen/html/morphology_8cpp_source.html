301 bool FileMorphology::GenerateFromHDF(int const N)
302 {
303  // First read in three individual axes, then put into positions,
304  // finally scale and transform into range [-1, 1].
305  std::vector<float> axes[4];
306  const char* coord_ids[4]{ "/x", "/y", "/z", "/region_id" };
307  int numpnts;
308  int n_dims = read_regions ? 4 : 3;
309  try
310  {
311  H5File file(fl_name, H5F_ACC_RDONLY);
312  for (int i = 0; i < n_dims; ++i)
313  {
314  DataSet axis = file.openDataSet(coord_ids[i]);
315  numpnts = (int)(axis.getStorageSize() / sizeof(float));
316  axes[i].resize(numpnts);
317  axis.read(axes[i].data(), PredType::NATIVE_FLOAT);
318  }
319  }
320  catch (...)
321  {
322  return false;
323  }
324 
325  // Calculate bounding box.
326  glm::vec3 wmin, wmax;
327  for (int i = 0; i < 3; ++i)
328  {
329  wmin[i] = *std::min_element(axes[i].begin(), axes[i].end());
330  wmax[i] = *std::max_element(axes[i].begin(), axes[i].end());
331  }
332 
333  // Check if we have enough points.
334  if (N > numpnts)
335  {
336  std::cerr << "Not enough points in file " << fl_name
337  << ". Have N = " << N << " and " << numpnts
338  << " points available." << std::endl;
339  return false;
340  }
341 
342  // Sample points uniformly so that we have exactly N, not numpnts.
343  int stride{ numpnts / N };
344  positions.resize(N);
345  for (int i = 0; i < N; ++i)
346  {
347  int const j{ stride * i };
348  positions[i] = { axes[0][j], axes[1][j], axes[2][j] };
349  }
350 
351  if (read_regions)
352  {
353  regions.resize(N);
354  for (int i = 0; i < N; ++i)
355  {
356  int const j{ stride * i };
357  regions[i] = axes[n_dims - 1][j];
358  }
359  std::set<int> unique_regions_set(regions.begin(), regions.end());
360  unique_regions = std::vector<int>(unique_regions_set.begin(), unique_regions_set.end());
361  }
362 
363  // Translate so that min is zero.
364  Translate(-wmin);
365 
366  // Scale so that everything is in the range [0, 2].
367  Scale(glm::vec3(2.0f) / (wmax - wmin));
368 
369  // Finally, translate so that min is -1 and everything is in range [-1, 1].
370  Translate(glm::vec3(-1));
371 
372  return true;
373 }
