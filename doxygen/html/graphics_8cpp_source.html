580 void write_frame(FILE* out, uint8_t* data)
581 {

583  glReadPixels(0, 0, w, h, GL_RGB, GL_UNSIGNED_BYTE, data);
584  // invert along y-axis

586  for (int x = 0; x < w; ++x)
587  for (int chan = 0; chan < 3; ++chan)
588  std::swap(data[3 * (y*w + x) + chan], data[3 * ((h - 1 - y)*w + x) + chan]);
589 
590  // Calculate size of file after it will be written to.
591  g_attrs.raw_video_frame_idx += 1;
592  size_t sz_in_bytes = w * h * 3 * sizeof(uint8_t) * g_attrs.raw_video_frame_idx;
593  int const sz_in_gb = (int)(sz_in_bytes / 1e9);
594  if (sz_in_gb >= g_attrs.max_raw_video_size_gb)
595  {
596  // Convert file to mov using batch file script.
597  system(("convert.bat " + std::to_string(g_attrs.mov_video_idx)).c_str());
598  g_attrs.mov_video_idx += 1;
599 
600  // Go back to start of raw data file.
601  fseek(out, 0, SEEK_SET);
602  g_attrs.raw_video_frame_idx = 0;
603  }
604 
605  fwrite(data, sizeof(uint8_t), w*h*3, out);
606 }
